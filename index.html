else if (action === "getCheckpoints") result = getCheckpoints(); else if (action === "getPatrolLogs") result = getPatrolLogs(); else if (action === "submitPatrolLog") result = submitPatrolLog(args[0]); else if (action === "getOfficers") result = getOfficers(); // Mengambil data Admin else if (action === "saveOfficer") result = saveOfficer(args[0]); // Menyimpan data Admin else if (action === "deleteOfficer") result = deleteOfficer(args[0]); // Menghapus data Admin else if (action === "saveCheckpoint") result = saveCheckpoint(args[0]); else if (action === "deleteCheckpoint") result = deleteCheckpoint(args[0]); else if (action === "saveSchedule") result = saveSchedule(args[0]); else if (action === "deleteSchedule") result = deleteSchedule(args[0]); else if (action === "setupDatabase") result = setupDatabase(); else throw new Error("Aksi POST tidak dikenal: " + action); return ContentService.createTextOutput(JSON.stringify({ success: true, data: result })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } /** * Mendapatkan atau menginisialisasi spreadsheet database */ function getDb() { if (SPREADSHEET_ID) { try { return SpreadsheetApp.openById(SPREADSHEET_ID); SPREADSHEET_ID = newSs.getId(); return newSs; } } /** * FUNGSI SETUP DATABASE AWAL (Hanya Admin) */ function setupDatabase() { var ss = getDb(); // 1. Sheet Schedules (Jadwal Patroli) var scheduleSheet = ss.getSheetByName("Schedules"); if (!scheduleSheet) { scheduleSheet = ss.insertSheet("Schedules"); scheduleSheet.appendRow(["ID Jadwal", "Nama Shift", "Jam Mulai", "Jam Selesai", "Keterangan"]); scheduleSheet.appendRow(["S1", "Shift Pagi", "08:00", "12:00", "Patroli berkala setiap 2 jam"]); scheduleSheet.appendRow(["S2", "Shift Siang", "13:00", "17:00", "Patroli berkala setiap 2 jam"]); scheduleSheet.appendRow(["S3", "Shift Malam", "20:00", "04:00", "Patroli ketat setiap 1 jam"]); } // 4. Sheet PatrolLogs (Log Tap RFID oleh Admin) var logSheet = ss.getSheetByName("PatrolLogs"); if (!logSheet) { logSheet = ss.insertSheet("PatrolLogs"); logSheet.appendRow(["Waktu Tap", "ID Admin", "Nama Admin", "UID RFID", "Nama Lokasi", "Status", "Catatan"]); } else { logSheet.getRange(1, 2).setValue("ID Admin"); logSheet.getRange(1, 3).setValue("Nama Admin"); } var defaultSheet = ss.getSheetByName("Sheet1"); if (defaultSheet) { try { ss.deleteSheet(defaultSheet); } catch(e) {} } return { success: true, spreadsheetUrl: ss.getUrl(), spreadsheetId: ss.getId() }; } /** * Autentikasi Pengguna (Hanya Admin) */ function loginUser(userId, pin) { var ss = getDb(); var sheet = ss.getSheetByName("Users"); if (!sheet) { return { success: false, message: "Database Admin belum diinisialisasi! Jalankan Setup terlebih dahulu." }; } var data = sheet.getDataRange().getValues(); if (userId === null || userId === undefined || pin === null || pin === undefined) { return { success: false, message: "ID Admin dan PIN tidak boleh kosong!" }; } var inputId = userId.toString().trim().toLowerCase(); var inputPin = pin.toString().trim().split('.')[0]; for (var i = 1; i < data.length; i++) { if (!data[i][0] || data[i][0].toString().trim() === "") continue; var dbId = data[i][0].toString().trim().toLowerCase(); var rawDbPin = data[i][2]; var dbPin = ""; if (typeof rawDbPin === "number") { dbPin = Math.round(rawDbPin).toString(); } else { dbPin = rawDbPin ? rawDbPin.toString().trim().split('.')[0] : ""; } if (dbId === inputId && dbPin === inputPin) { return { success: true, userId: data[i][0].toString(), name: data[i][1].toString(), role: "Admin" // Semua dipastikan login sebagai Admin }; } } return { success: false, message: "ID Admin atau PIN salah!" }; } /** * Mendapatkan Daftar Admin */ function getOfficers() { var ss = getDb(); var sheet = ss.getSheetByName("Users"); if (!sheet) return []; var data = sheet.getDataRange().getValues(); var officers = []; for (var i = 1; i < data.length; i++) { if (!data[i][0] || data[i][0].toString().trim() === "") continue; var rawPin = data[i][2]; var formattedPin = ""; if (rawPin !== null && rawPin !== undefined) { formattedPin = typeof rawPin === "number" ? Math.round(rawPin).toString() : rawPin.toString().trim().split('.')[0]; } officers.push({ id: safeString(data[i][0]), name: safeString(data[i][1]), pin: formattedPin, role: "Admin" // Tetap paksa sebagai Admin }); } return officers; } /** * Tambah atau Update Admin */ function saveOfficer(officer) { var ss = getDb(); var sheet = ss.getSheetByName("Users"); var data = sheet.getDataRange().getValues(); var foundIndex = -1; var targetId = officer.id.toString().trim().toLowerCase(); for (var i = 1; i < data.length; i++) { if (!data[i][0] || data[i][0].toString().trim() === "") continue; if (data[i][0].toString().trim().toLowerCase() === targetId) { foundIndex = i + 1; break; } } var cleanPin = officer.pin.toString().trim().split('.')[0]; if (foundIndex > -1) { sheet.getRange(foundIndex, 2).setValue(officer.name); sheet.getRange(foundIndex, 3).setValue("'" + cleanPin); sheet.getRange(foundIndex, 4).setValue("Admin"); // Set default role sebagai Admin } else { sheet.appendRow([officer.id, officer.name, "'" + cleanPin, "Admin"]); } return { success: true }; } /** * Hapus Admin */ function deleteOfficer(id) { var ss = getDb(); var sheet = ss.getSheetByName("Users"); var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (!data[i][0] || data[i][0].toString().trim() === "") continue; if (data[i][0].toString().trim().toLowerCase() === id.toString().trim().toLowerCase()) { sheet.deleteRow(i + 1); return { success: true }; } } return { success: false, message: "Admin tidak ditemukan" }; } /** * Mendapatkan Daftar Checkpoint (RFID) */ function getCheckpoints() { var ss = getDb(); var sheet = ss.getSheetByName("Checkpoints"); if (!sheet) return []; var data = sheet.getDataRange().getValues(); var checkpoints = []; for (var i = 1; i < data.length; i++) { if (!data[i][0] || data[i][0].toString().trim() === "") continue; checkpoints.push({ uid: safeString(data[i][0]), name: safeString(data[i][1]), description: safeString(data[i][2]) }); } return checkpoints; } /** * Tambah atau Update Checkpoint RFID */ function saveCheckpoint(checkpoint) { var ss = getDb(); var sheet = ss.getSheetByName("Checkpoints"); var data = sheet.getDataRange().getValues(); var foundIndex = -1; for (var i = 1; i < data.length; i++) { if (!data[i][0] || data[i][0].toString().trim() === "") continue; if (data[i][0].toString().toLowerCase() === checkpoint.uid.toLowerCase()) { foundIndex = i + 1; break; } } if (foundIndex > -1) { sheet.getRange(foundIndex, 2).setValue(checkpoint.name); sheet.getRange(foundIndex, 3).setValue(checkpoint.description); } else { sheet.appendRow([checkpoint.uid.toLowerCase(), checkpoint.name, checkpoint.description]); } return { success: true }; } /** * Hapus Checkpoint */ function deleteCheckpoint(uid) { var ss = getDb(); var sheet = ss.getSheetByName("Checkpoints"); var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (!data[i][0] || data[i][0].toString().trim() === "") continue; if (data[i][0].toString().toLowerCase() === uid.toLowerCase()) { sheet.deleteRow(i + 1); return { success: true }; } } return { success: false, message: "Checkpoint tidak ditemukan" }; } /** * Mendapatkan Jadwal Patroli */ function getSchedules() { var ss = getDb(); var sheet = ss.getSheetByName("Schedules"); if (!sheet) return []; var data = sheet.getDataRange().getValues(); var schedules = []; for (var i = 1; i < data.length; i++) { if (!data[i][0] || data[i][0].toString().trim() === "") continue; schedules.push({ id: safeString(data[i][0]), name: safeString(data[i][1]), startTime: safeString(data[i][2]), endTime: safeString(data[i][3]), description: safeString(data[i][4]) }); } return schedules; } /** * Tambah atau Update Jadwal */ function saveSchedule(schedule) { var ss = getDb(); var sheet = ss.getSheetByName("Schedules"); var data = sheet.getDataRange().getValues(); var foundIndex = -1; for (var i = 1; i < data.length; i++) { if (!data[i][0] || data[i][0].toString().trim() === "") continue; if (data[i][0].toString().trim().toLowerCase() === schedule.id.toString().trim().toLowerCase()) { foundIndex = i + 1; break; } } if (foundIndex > -1) { sheet.getRange(foundIndex, 2).setValue(schedule.name); sheet.getRange(foundIndex, 3).setValue(schedule.startTime); sheet.getRange(foundIndex, 4).setValue(schedule.endTime); sheet.getRange(foundIndex, 5).setValue(schedule.description); } else { sheet.appendRow([schedule.id, schedule.name, schedule.startTime, schedule.endTime, schedule.description]); } return { success: true }; } /** * Hapus Jadwal */ function deleteSchedule(id) { var ss = getDb(); var sheet = ss.getSheetByName("Schedules"); var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (!data[i][0] || data[i][0].toString().trim() === "") continue; if (data[i][0].toString().trim().toLowerCase() === id.toString().trim().toLowerCase()) { sheet.deleteRow(i + 1); return { success: true }; } } return { success: false, message: "Jadwal tidak ditemukan" }; } /** * Menulis Hasil Tap / Scan Patroli oleh Admin ke Log Spreadsheet */ function submitPatrolLog(logData) { var ss = getDb(); var logSheet = ss.getSheetByName("PatrolLogs"); var checkpointSheet = ss.getSheetByName("Checkpoints"); var userSheet = ss.getSheetByName("Users"); var checkpointData = checkpointSheet.getDataRange().getValues(); var userData = userSheet.getDataRange().getValues(); var locationName = "Titik Tidak Terdaftar"; var adminName = "Admin Tidak Dikenal"; // Cari nama lokasi berdasarkan RFID UID for (var i = 1; i < checkpointData.length; i++) { if (!checkpointData[i][0] || checkpointData[i][0].toString().trim() === "") continue; if (checkpointData[i][0].toString().toLowerCase() === logData.uid.toLowerCase()) { locationName = checkpointData[i][1]; break; } } // Cari nama Admin berdasarkan ID yang sedang login for (var j = 1; j < userData.length; j++) { if (!userData[j][0] || userData[j][0].toString().trim() === "") continue; if (userData[j][0].toString().toLowerCase() === logData.officerId.toString().toLowerCase()) { adminName = userData[j][1]; break; } } // Evaluasi Waktu Patroli var status = "Sesuai Jadwal"; if (logData.scheduleId) { var schedules = getSchedules(); var activeSched = schedules.find(function(s) { return s.id === logData.scheduleId; }); if (activeSched) { var now = new Date(); var currentHourMin = Utilities.formatDate(now, Session.getScriptTimeZone(), "HH:mm"); if (currentHourMin < activeSched.startTime || currentHourMin > activeSched.endTime) { status = "Luar Jam Dinas (" + activeSched.name + ")"; } } } var timestamp = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss"); // Tulis baris baru ke log logSheet.appendRow([ timestamp, logData.officerId, adminName, logData.uid.toLowerCase(), locationName, status, logData.notes || "-" ]); return { success: true, timestamp: timestamp, location: locationName, officer: adminName, status: status }; } /** * Mendapatkan Semua Log Patroli (Batas maks 500 baris terakhir untuk performa) */ function getPatrolLogs() { var ss = getDb(); var sheet = ss.getSheetByName("PatrolLogs"); if (!sheet) return []; var data = sheet.getDataRange().getValues(); var logs = []; var startRow = Math.max(1, data.length - 500); for (var i = data.length - 1; i >= startRow; i--) { if (!data[i][0] || data[i][0].toString().trim() === "") continue; logs.push({ timestamp: safeTimestamp(data[i][0]), officerId: safeString(data[i][1]), officerName: safeString(data[i][2]), uid: safeString(data[i][3]), location: safeString(data[i][4]), status: safeString(data[i][5]), notes: safeString(data[i][6]) }); } return logs; } /** * Helper: Konversi data string aman */ function safeString(val) { if (val === null || val === undefined) return ""; if (val instanceof Date) { try { return Utilities.formatDate(val, Session.getScriptTimeZone(), "HH:mm"); } catch(e) { return val.toString(); } } return val.toString().trim(); } /** * Helper: Konversi timestamp tanggal secara aman */ function safeTimestamp(val) { if (val === null || val === undefined || val === "") return ""; if (val instanceof Date) { try { return Utilities.formatDate(val, Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss"); } catch(e) { return val.toString(); } } return val.toString().trim(); }
